    <script>
        const initThreeJS = () => {
            const container = document.getElementById('three-canvas-container');
            const scene = new THREE.Scene();

            const aspect = window.innerWidth / window.innerHeight;
            const camera = new THREE.OrthographicCamera(-aspect * 10, aspect * 10, 10, -10, 0.1, 100);
            camera.position.z = 10;

            const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
            renderer.setSize(window.innerWidth, window.innerHeight);
            container.appendChild(renderer.domElement);

            const gridGroup = new THREE.Group();
            const gridMat = new THREE.LineBasicMaterial({ color: '#1e3a8a', transparent: true, opacity: 0.15 });
            for (let i = -50; i <= 50; i += 1.5) {
                gridGroup.add(new THREE.Line(new THREE.BufferGeometry().setFromPoints([new THREE.Vector3(i, -50, 0), new THREE.Vector3(i, 50, 0)]), gridMat));
                gridGroup.add(new THREE.Line(new THREE.BufferGeometry().setFromPoints([new THREE.Vector3(-50, i, 0), new THREE.Vector3(50, i, 0)]), gridMat));
            }
            scene.add(gridGroup);

            const createGear = (radius, teeth, innerRadius, color) => {
                const pts = [];
                for(let i=0; i<=teeth * 2; i++) {
                    const a = (i / (teeth * 2)) * Math.PI * 2;
                    const r = i % 2 === 0 ? radius : innerRadius;
                    pts.push(new THREE.Vector3(Math.cos(a)*r, Math.sin(a)*r, 0));
                }
                return new THREE.LineLoop(new THREE.BufferGeometry().setFromPoints(pts), new THREE.LineBasicMaterial({ color, transparent: true, opacity: 0.2 }));
            };

            const gearMain = createGear(5, 32, 4.2, '#38bdf8');
            const gearSub = createGear(2.5, 16, 2.0, '#818cf8');
            gearMain.position.set(-8, -3, 0); gearSub.position.set(-1.8, 1.2, 0);
            scene.add(gearMain); scene.add(gearSub);

            let mouseX = 0; let mouseY = 0;
            document.addEventListener('mousemove', (e) => {
                mouseX = (e.clientX / window.innerWidth) * 2 - 1;
                mouseY = -(e.clientY / window.innerHeight) * 2 + 1;
            });

            const animate = () => {
                requestAnimationFrame(animate);
                gridGroup.position.x += (mouseX * 0.5 - gridGroup.position.x) * 0.05;
                gridGroup.position.y += (mouseY * 0.5 - gridGroup.position.y) * 0.05;
                gearMain.rotation.z -= 0.005;
                gearSub.rotation.z += 0.01; 
                renderer.render(scene, camera);
            };
            animate();

            window.addEventListener('resize', () => {
                const aspect = window.innerWidth / window.innerHeight;
                camera.left = -aspect * 10; camera.right = aspect * 10;
                camera.updateProjectionMatrix();
                renderer.setSize(window.innerWidth, window.innerHeight);
            });
        };
        initThreeJS();
    </script>
